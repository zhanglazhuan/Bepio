idf_component_register(SRCS "bepio_tts.c" "tts_text.c" INCLUDE_DIRS "."
                       PRIV_REQUIRES esp-sr esp_partition mbedtls esp_timer)

# Voice data is separate from the app; regular `idf.py flash` writes both.
idf_build_get_property(components BUILD_COMPONENTS)
if(esp-sr IN_LIST components)
    set(sr_component esp-sr)
else()
    set(sr_component espressif__esp-sr)
endif()
idf_component_get_property(sr_path ${sr_component} COMPONENT_DIR)
set(voice_image "${sr_path}/esp-tts/esp_tts_chinese/esp_tts_voice_data_xiaoxin_small.dat")
if(NOT EXISTS "${voice_image}")
    message(FATAL_ERROR "ESP-TTS Xiaoxin voice data missing: ${voice_image}")
endif()
file(SIZE "${voice_image}" VOICE_BYTES)
file(SHA256 "${voice_image}" voice_sha)
string(REGEX MATCHALL ".." voice_sha_bytes "${voice_sha}")
list(TRANSFORM voice_sha_bytes PREPEND "0x")
list(JOIN voice_sha_bytes ", " VOICE_SHA256_BYTES)
configure_file(voice_model.h.in "${CMAKE_CURRENT_BINARY_DIR}/voice_model.h" @ONLY)
target_include_directories(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")

partition_table_get_partition_info(voice_size "--partition-name voice_data" "size")
if(NOT voice_size)
    message(FATAL_ERROR "ESP-TTS needs a voice_data data partition (at least ${VOICE_BYTES} bytes)")
endif()
if(VOICE_BYTES GREATER voice_size)
    message(FATAL_ERROR "voice_data partition too small for Xiaoxin voice")
endif()
add_custom_target(bepio_tts_voice ALL DEPENDS "${voice_image}")
add_dependencies(flash bepio_tts_voice)
esptool_py_flash_to_partition(flash "voice_data" "${voice_image}")
